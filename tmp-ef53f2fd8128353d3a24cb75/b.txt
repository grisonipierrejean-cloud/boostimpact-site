-- RENOM, webhooks de base de données vers n8n, version SANS secret dans le SQL (16/10/2026). Remplace webhooks.sql.
-- Le secret partagé (en-tête X-Renom-Secret, le même que l'identifiant n8n « secret des webhooks ») est rangé dans le coffre de Supabase
-- (Vault, chiffré) sous le nom renom_webhook_secret ; ce fichier ne le contient pas et peut donc être collé tel quel, et rejoué.
-- Adresse n8n : peejay.app.n8n.cloud (à changer ici seulement si l'instance change).
-- Corps envoyé : même forme que les webhooks de base de données de Supabase (type, table, schema, record, old_record), lue par les workflows n8n.
-- Envoi par pg_net (asynchrone) : un n8n en panne ne bloque jamais l'écriture dans la base. Sans secret dans le coffre : aucun envoi.
create extension if not exists pg_net with schema extensions;

create or replace function public.renom_hook() returns trigger
language plpgsql security definer set search_path = public, extensions, vault as $$
declare s text;
begin
  select decrypted_secret into s from vault.decrypted_secrets where name = 'renom_webhook_secret' limit 1;
  if s is null or s = '' then return null; end if;
  perform net.http_post(
    url := 'https://peejay.app.n8n.cloud/webhook/' || tg_argv[0],
    body := jsonb_build_object('type', tg_op, 'table', tg_table_name, 'schema', tg_table_schema,
                               'record', case when tg_op = 'DELETE' then null else to_jsonb(new) end,
                               'old_record', case when tg_op = 'INSERT' then null else to_jsonb(old) end),
    headers := jsonb_build_object('Content-Type', 'application/json', 'X-Renom-Secret', s),
    timeout_milliseconds := 5000);
  return null;
end $$;
revoke all on function public.renom_hook() from public, anon, authenticated;
