-- ---------------------------------------------------------------- 0 bis. Référence courte de l'avis (rattachement sûr des réponses WhatsApp)
-- Une lettre, un chiffre, une lettre (« A7K ») : sans I ni O, sans 0 ni 1 (jamais confondus), toujours un chiffre (jamais un mot comme « LES »).
-- Unique par commerce (4 608 possibilités ; au-delà de 200 essais, un chiffre de plus). Posée à l'arrivée de chaque avis ; rattrapage des avis en file.
-- Contre-revue (G) : posée APRÈS l'insertion, dans la même transaction, sous un verrou par commerce, avec une nouvelle tentative si la contrainte
-- d'unicité est violée (deux avis du même commerce arrivés en même temps) : l'insertion de l'avis n'échoue jamais (au pire, avis sans référence :
-- le rattachement sans référence s'applique, l'équipe est alertée par la liste « lequel ? »).
create or replace function nouvelle_ref_avis(p_commerce uuid) returns text language plpgsql volatile set search_path = public as $$
declare l text := 'ABCDEFGHJKLMNPQRSTUVWXYZ'; c text := '23456789'; r text; i int := 0;
begin
  loop
    r := substr(l, 1 + floor(random() * 24)::int, 1) || substr(c, 1 + floor(random() * 8)::int, 1) || substr(l, 1 + floor(random() * 24)::int, 1);
    if i > 200 then r := r || substr(c, 1 + floor(random() * 8)::int, 1); end if;
    exit when not exists (select 1 from avis where commerce_id = p_commerce and ref_courte = r);
    i := i + 1;
  end loop;
  return r;
end $$;
revoke all on function nouvelle_ref_avis(uuid) from public, anon, authenticated;
create or replace function attribuer_ref_avis(p_avis uuid, p_commerce uuid) returns text language plpgsql security definer set search_path = public as $$
declare i int := 0; r text;
begin
  perform pg_advisory_xact_lock(hashtext('renom_ref_avis'), hashtext(coalesce(p_commerce::text, '')));
  loop
    begin
      r := nouvelle_ref_avis(p_commerce);
      update avis set ref_courte = r where id = p_avis;
      return r;
    exception when unique_violation then
      i := i + 1;
      if i >= 20 then return null; end if;   -- jamais d'échec : l'avis reste sans référence
    end;
  end loop;
end $$;
revoke all on function attribuer_ref_avis(uuid, uuid) from public, anon, authenticated;
create or replace function poser_ref_avis() returns trigger language plpgsql security definer set search_path = public as $$
begin
  if new.ref_courte is null then perform attribuer_ref_avis(new.id, new.commerce_id); end if;
  return null;
end $$;
revoke all on function poser_ref_avis() from public, anon, authenticated;
drop trigger if exists poser_ref_avis on avis;
create trigger poser_ref_avis after insert on avis for each row when (new.ref_courte is null) execute function poser_ref_avis();
do $l11r$
declare x record;
begin
  for x in select id, commerce_id from avis where ref_courte is null and statut in ('nouveau','a_valider','valide','escalade','humain','a_publier_main') loop
    perform attribuer_ref_avis(x.id, x.commerce_id);
  end loop;
end $l11r$;
-- Références lues dans un message : jetons lettre-chiffre-lettre (un chiffre de plus permis), en capitales. Seules celles d'un avis du commerce comptent.
create or replace function refs_du_message(p_texte text) returns text[] language sql immutable as $$
  select coalesce(array_agg(distinct upper(m[1])), '{}') from regexp_matches(coalesce(p_texte, ''), '\m([A-Za-z][2-9][A-Za-z][2-9]?)\M', 'g') m $$;

-- Contre-revue (B) : lecture d'un message (référence déjà retirée). Le préfixe « Réf. » et la ponctuation autour des mots sont retirés, puis OK,
-- ATTENDS ou FAUX (WAIT, WACHT ; FAKE, VALS) sont cherchés SEULS, avec au plus deux mots de politesse au début ou à la fin (svp, merci, stp…).
-- Contre-revue n°2 : un message qui contient une négation ou un mot de refus (« non », « ne publiez pas », « pas maintenant », « stop »,
-- « attendez », « supprimez », « don't », « not now », « niet », « nee »…) va à l'équipe ('a_lire') ; une correction n'est retenue que si elle
-- propose un contenu : au moins trois mots qui ne sont ni un mot-clé, ni de la politesse, ni un refus. Dans le doute : 'a_lire'.
-- Rend 'ok', 'attends', 'faux', 'a_lire', 'correction', ou null pour un message vide.
create or replace function lire_message_whatsapp(p_texte text) returns text language plpgsql immutable set search_path = public as $$
declare s text := lower(coalesce(p_texte, '')); w text[]; n int := 0; k text; contenu int;
  pol text[] := array['svp','stp','merci','beaucoup','please','pls','thanks','thx','bedankt','aub','alstublieft','bonjour','bonsoir','hello','hi','dag','cordialement'];
  cle text[] := array['ok','okay','oké','oke','okey','oui','yes','ja','daccord','akkoord','attends','wait','wacht','faux','fake','vals','nep'];
  refus text[] := array['non','ne','n','pas','jamais','aucun','aucune','rien','no','not','dont','don','never','nothing','nee','niet','geen','nooit','niks',
    'stop','stoppez','stoppen','attendez','attendre','attente','patientez','supprimez','supprimer','supprime','effacez','effacer','retirez','retirer',
    'annulez','annuler','delete','remove','cancel','hold','later','verwijder','verwijderen','annuleer','wachten','straks',
    -- contre-revue n°3 : report, appel, « pas un client », accord implicite (« c'est bon », « parfait » : jamais un OK automatique)
    'tard','verra','tomber','gardez','garder','appelle','appelez','appeler','rappelez','rappeler','telephone','téléphone',
    'fermé','fermés','fermée','fermées','concurrent','concurrents','concurrente','concurrence',
    'closed','competitor','competitors','gesloten','dicht','voorlopig','concurrentie','concurrenten'];
  -- Mots seuls (« parfait », « super », « bon ») : déjà « à lire » (pas de contenu) ; formules d'accord ou de report sur plusieurs mots :
  phrases text := '\m(laissez tomber|on verra|plus tard|pour le moment|pour l instant|je vous appelle|appelez moi|rappelez moi|jamais venus?|jamais venues?|faux client|pas client|'
    || 'c est bon|ca me va|ça me va|tres bien|très bien|c est parfait|'
    || 'let it go|forget it|we ll see|for now|not yet|keep it|call me|i ll call|never came|never been|not a customer|fake customer|it s fine|sounds good|all good|'
    || 'laat maar|laat het|we zien wel|voor nu|hou het|houd het|bel me|ik bel|nooit geweest|nooit gekomen|geen klant|is goed|prima zo)\M';
begin
  s := replace(replace(s, chr(8217), ''''), 'd''accord', 'daccord');
  s := regexp_replace(s, '[[:punct:]«»“”„…–—' || chr(8216) || ']', ' ', 'g');
  s := btrim(regexp_replace(s, '\s+', ' ', 'g'));
  if s = '' then return null; end if;
  s := regexp_replace(s, '\m(r[ée]f[ée]rence|r[ée]f|ref|reference|referentie)\M', ' ', 'g');
  s := regexp_replace(s, '\ms il (vous|te) pla[iî]t\M|\mdank (u|je)( wel)?\M|\mthank you\M|\malsjeblieft\M', ' svp ', 'g');
  s := regexp_replace(s, '\m(faux avis|fake review|nep review|valse review)\M', ' faux ', 'g');
  s := btrim(regexp_replace(s, '\s+', ' ', 'g'));
  if s = '' then return null; end if;   -- « Réf. » seul : rien à lire (la référence seule est redemandée)
  w := string_to_array(s, ' ');
  if w <@ pol then return 'a_lire'; end if;   -- « merci » seul : l'équipe lit, ce n'est pas une correction
  while n < 2 and cardinality(w) > 1 and (w[1] = any(pol) or w[cardinality(w)] = any(pol)) loop
    if w[1] = any(pol) then w := w[2:]; else w := w[1:cardinality(w) - 1]; end if;
    n := n + 1;
  end loop;
  if cardinality(w) = 1 then
    k := w[1];
    if k in ('ok','okay','oké','oke','okey','oui','yes','ja','daccord','akkoord') then return 'ok'; end if;
    if k in ('attends','wait','wacht') then return 'attends'; end if;
    if k in ('faux','fake','vals','nep') then return 'faux'; end if;
  end if;
  w := string_to_array(s, ' ');
  if w && refus or w && cle or s ~ '(faux|attends|wacht)' or s ~ phrases then return 'a_lire'; end if;
  select count(*) into contenu from unnest(w) x where not (x = any(pol));
  if contenu < 3 then return 'a_lire'; end if;
  return 'correction';
end $$;
revoke all on function lire_message_whatsapp(text) from public, anon, authenticated;
-- Principe (contre-revues) : DANS LE DOUTE, ON NE PUBLIE PAS. Contre-revue n°2 : la suspension vise les avis EN COURS au moment du message
-- (ceux de la liste « lequel ? »), jamais les avis arrivés après ; elle n'est pas un ATTENDS (retenues_sans_message ne publie jamais un avis
-- suspendu) ; fermer la tâche ou « Message lu » ne la lève pas : seulement « Appliquer OK / ATTENDS / FAUX » (admin_appliquer_reponse) ou un
-- nouvel envoi (« Envoyer au commerçant »). Un message sans mot-clé alors qu'AUCUN avis n'est en cours (« Merci », une photo) ne suspend rien.
drop function if exists message_commercant_en_suspens(uuid);
create or replace function suspendre_avis(p_ids uuid[], p_motif text) returns int language plpgsql security definer set search_path = public as $$
declare k int;
begin
  update avis set suspendu_le = coalesce(suspendu_le, now()), suspendu_motif = coalesce(suspendu_motif, p_motif), api_demande_le = null
   where id = any(coalesce(p_ids, '{}')) and statut in ('escalade','humain');
  get diagnostics k = row_count;
  return k;
end $$;
revoke all on function suspendre_avis(uuid[], text) from public, anon, authenticated;
-- Historique des suspensions (comme avis_retenues) : ouverte à la suspension, close à la levée ; l'alerte « bientôt dépassé » repart.
create or replace function avis_suspension_historique() returns trigger language plpgsql security definer set search_path = public as $$
begin
  if new.suspendu_le is not null and old.suspendu_le is null then
    insert into avis_suspensions (avis_id, commerce_id, suspendu_le, motif) values (new.id, new.commerce_id, new.suspendu_le, new.suspendu_motif) on conflict do nothing;
  elsif new.suspendu_le is null and old.suspendu_le is not null then
    update avis_suspensions set levee_le = greatest(now(), suspendu_le) where avis_id = new.id and levee_le is null;
    update avis set alerte_delai_le = null where id = new.id;
  end if;
  return null;
end $$;
revoke all on function avis_suspension_historique() from public, anon, authenticated;
drop trigger if exists avis_suspension_historique on avis;
create trigger avis_suspension_historique after update of suspendu_le on avis for each row execute function avis_suspension_historique();
-- Échéance d'un avis (CGV art. 9 et 10 ; corps du bloc « Attendez mon appel » recopié) : les suspensions comptent comme les retenues.
create or replace function echeance_avis(p_avis uuid, p_recu timestamptz) returns timestamptz language plpgsql stable security definer set search_path = public as $$
declare h interval := make_interval(hours => (conditions_offre()->'garantie_delais'->>'avis_traite_heures')::int); e timestamptz := p_recu + h; r record;
begin
  if p_recu is null then return null; end if;
  for r in select retenue_le debut, liberee_le fin from avis_retenues where avis_id = p_avis
           union all select suspendu_le, levee_le from avis_suspensions where avis_id = p_avis order by 1 loop
    if r.debut >= e then continue; end if;          -- posée après l'échéance : le retard était déjà là
    if r.fin is null then return null; end if;      -- en cours : délai suspendu, aucune échéance
    e := greatest(e, r.fin + h);                    -- levée : le délai repart pour 24 h
  end loop;
  return e;
end $$;
revoke all on function echeance_avis(uuid, timestamptz) from public, anon, authenticated;
grant execute on function echeance_avis(uuid, timestamptz) to service_role;

-- ---------------------------------------------------------------- 0. Alertes au commerçant 24 h sur 24 (décision finale de PJ du 17/11/2026)
-- Plus aucune pause la nuit : les alertes au commerçant partent tout de suite, à toute heure. Paramètre gardé à 0 (debut = fin : aucune pause) ;
-- REGLES_WHATSAPP de moteur/courriels/whatsapp.js porte la même valeur (égalité testée).
create or replace function heures_calmes_commercant() returns jsonb language sql stable as $$ select '{"debut": 0, "fin": 0, "fuseau": "Europe/Paris"}'::jsonb $$;
create or replace function whatsapp_regles() returns jsonb language sql stable as $$
  select '{"ecart_min_secondes": 30, "alea_max_secondes": 15, "max_par_heure": 20, "attente_max_heures": 2, "fenetre_secondes": 300, "max_par_passage": 10}'::jsonb
         || jsonb_build_object('heures_calmes', heures_calmes_commercant()) $$;
-- Rend l'heure d'envoi permise la plus proche. debut = fin : aucune pause. Pause de nuit (debut > fin) ou de jour (debut < fin).
create or replace function apres_heures_calmes(p timestamptz) returns timestamptz language plpgsql stable as $$
declare r jsonb := whatsapp_regles()->'heures_calmes'; l timestamp := p at time zone (r->>'fuseau'); h int := extract(hour from l)::int;
  d int := (r->>'debut')::int; f int := (r->>'fin')::int;
begin
  if d = f then return p; end if;
  if d > f then
    if h >= d then return ((l::date + 1) + make_time(f, 0, 0)) at time zone (r->>'fuseau'); end if;
    if h < f then return (l::date + make_time(f, 0, 0)) at time zone (r->>'fuseau'); end if;
  elsif h >= d and h < f then return (l::date + make_time(f, 0, 0)) at time zone (r->>'fuseau');
  end if;
  return p;
end $$;

-- ---------------------------------------------------------------- 1. Heure de publication annoncée (CGV art. 9 et 10)
-- Une seule source pour la base et les tests : 6 h après le départ réel de l'alerte, raccourcie pour que la publication tombe au plus tard
-- 23 h après l'arrivée de l'avis (24 h de l'article 10, moins 1 h pour le passage de publication), jamais moins de 4 h pour réagir.
-- Alertes 24 h sur 24 : une alerte automatique part dès l'arrivée de l'avis, publication annoncée à reçu + 6 h : les 6 h tiennent toujours
-- (elles tiendraient même avec une pause de nuit de 12 h). Seule une réponse renvoyée tard (correction, réponse rédigée par l'équipe
-- plus de 19 h après l'arrivée) passe sous 6 h, jamais sous 4 h : si 4 h dépassent l'échéance, l'heure annoncée la dépasse (journal
-- AVIS_PUBLIE, dans_les_24h ; pendant une demande d'attendre, le délai est suspendu, article 10).
create or replace function avis_delais_annonce() returns jsonb language sql immutable as $$
  select '{"normal_heures": 6, "min_heures": 4, "marge_heures": 1}'::jsonb $$;
create or replace function avis_heure_annoncee(p_envoi timestamptz, p_recu timestamptz) returns timestamptz language sql stable as $$
  select greatest(p_envoi + make_interval(hours => (d->>'min_heures')::int),
                  least(p_envoi + make_interval(hours => (d->>'normal_heures')::int),
                        coalesce(p_recu, p_envoi) + make_interval(hours => 24 - (d->>'marge_heures')::int)))
    from (select avis_delais_annonce() d) x $$;

-- 2. Alerte à l'arrivée (corps du lot 10, recopié) : même règle d'heure pour TOUS les avis, sensibles compris (aucune pause : bloc 0) ; avis sensible : jamais de brouillon dans
-- l'alerte (le brouillon de Claude attend la relecture de l'équipe) ; heure de publication provisoire, recalculée au départ réel (déclencheur
-- annoncer_heure_publication).
create or replace function mettre_alerte_avis() returns trigger language plpgsql security definer set search_path = public as $$
declare c commerces; o offres; ty text; wa boolean; ap timestamptz := now(); src avis := new; pa timestamptz;
begin
  if tg_op = 'UPDATE' then
    if new.statut is not distinct from old.statut then return null; end if;
    src := old;
  end if;
  select * into c from commerces where id = new.commerce_id;
  select * into o from offres where id = c.offre_id;
  wa := whatsapp_actif(c);
  if new.statut = 'humain' or (new.statut = 'escalade' and not coalesce(new.sensible, false) and coalesce(new.note, 5) between 1 and 3
        and (tg_op = 'INSERT' or old.statut is distinct from 'humain') and new.traite_le is null) then
    ty := case when new.statut = 'humain' then 'avis_sensible' else 'avis_recu' end;
    if exists (select 1 from alertes_whatsapp where avis_id = new.id and type = ty) then return null; end if;   -- avis relu : pas de doublon
    if not wa then
      if not exists (select 1 from alertes_exploitant where type = 'ALERTE_AVIS_SANS_WHATSAPP' and detail->>'avis' = new.id::text) then
        insert into alertes_exploitant (type, commerce_id, detail) values ('ALERTE_AVIS_SANS_WHATSAPP', new.commerce_id, jsonb_build_object('avis', new.id, 'note', new.note));
        insert into journal (commerce_id, type, detail) values (new.commerce_id, 'ALERTE_AVIS_SANS_WHATSAPP', jsonb_build_object('avis', new.id, 'note', new.note));
      end if;
      return null;
    end if;
    -- Lot 11 : même règle pour toutes les alertes au commerçant, avis sensibles compris (apres_heures_calmes : aucune pause, bloc 0).
    ap := apres_heures_calmes(now());
    pa := case when ty = 'avis_recu' and nullif(btrim(coalesce(new.reponse_proposee, '')), '') is not null then avis_heure_annoncee(ap, new.cree_le) end;
    update avis set publication_pas_avant = coalesce(pa, avis_heure_annoncee(ap, new.cree_le)) where id = new.id and publication_pas_avant is null;
    if ty = 'avis_sensible' then update avis set retenue_le = coalesce(retenue_le, now()) where id = new.id and retenue_le is null; end if;
  elsif new.statut = 'publie' and tg_op = 'UPDATE' and old.statut in ('escalade','humain') and (o.avis_equipe or o.delegation) then
    ty := 'avis_traite';
    if wa then ap := apres_heures_calmes(now());
    elsif not (coalesce((c.notifications->>'email')::boolean, true) and email_valide(lower(btrim(c.email_validation)))) then return null;
    end if;
  else
    return null;
  end if;
  insert into alertes_whatsapp (commerce_id, avis_id, type, canal, prioritaire, envoyer_apres, donnees)
    values (new.commerce_id, new.id, ty, case when wa then 'whatsapp' else 'email' end, ty in ('avis_sensible','avis_recu'), ap,
            jsonb_build_object('note', coalesce(new.note, src.note), 'auteur', nullif(auteur_reduit(coalesce(new.auteur, src.auteur)), ''),
                               'reponse', case when ty = 'avis_traite' then left(coalesce(new.reponse, new.reponse_proposee), 2000)
                                               when ty = 'avis_recu' then left(new.reponse_proposee, 4096) end,   -- avis sensible : aucun brouillon
                               'texte', case when ty <> 'avis_traite' then left(new.texte, 4096) end, 'pas_avant', pa,
                               'ecrit_le', coalesce(new.ecrit_le, src.ecrit_le),
                               'echeance', case when ty <> 'avis_traite' then new.cree_le + interval '24 hours' end, 'sensible', ty = 'avis_sensible',
                               'auto', ty = 'avis_traite' and new.publie_par = 'api'))   -- revue : publiée par l'API, pas « par notre équipe »
    on conflict (avis_id, type) do nothing;
  return null;
end $$;
drop trigger if exists alerte_avis_commercant on avis;
create trigger alerte_avis_commercant after insert or update of statut on avis for each row execute function mettre_alerte_avis();

-- 3. La réponse proposée envoyée au commerçant (console : « Envoyer au commerçant ») : corps du lot 10, même règle d'heure pour tous, heure provisoire.
create or replace function admin_avis_proposer(p_id uuid, p_reponse text) returns jsonb language plpgsql security definer set search_path = public as $$
declare a avis; t text := nullif(btrim(coalesce(p_reponse, '')), ''); c commerces; wa boolean; pa timestamptz; ap timestamptz;
begin
  if not est_admin() then raise exception 'réservé à l''administrateur' using errcode = 'P0005'; end if;
  select * into a from avis where id = p_id for update;
  if a.id is null or a.statut not in ('escalade','humain') then raise exception 'Cet avis n''est pas dans la file de BoostImpact.' using errcode = 'P0061'; end if;
  if a.faux_signale_le is not null then
    raise exception 'Avis signalé FAUX par le commerçant : « Débloquer » d''abord (après vérification avec lui).' using errcode = 'P0061'; end if;
  if t is null then raise exception 'Écrivez la réponse proposée.' using errcode = 'P0061'; end if;
  if char_length(t) > 2000 then raise exception 'La réponse est trop longue (2 000 caractères au plus).' using errcode = 'P0061'; end if;
  if exists (select 1 from alertes_whatsapp where avis_id = a.id and type = 'avis_reponse' and statut = 'en_cours') then
    raise exception 'Un message sur cet avis part en ce moment : réessayez dans une minute.' using errcode = 'P0061'; end if;
  select * into c from commerces where id = a.commerce_id;
  wa := whatsapp_actif(c);
  update avis set reponse_proposee = t, propose_le = now(), traite_le = coalesce(traite_le, now()), api_demande_le = null where id = a.id;
  if wa then
    ap := apres_heures_calmes(now());   -- lot 11 : avis sensible compris
    if not (coalesce(a.sensible, false) or a.statut = 'humain') then
      pa := avis_heure_annoncee(ap, a.cree_le);   -- provisoire : recalculée au départ réel de l'alerte
      update avis set publication_pas_avant = pa where id = a.id;
    end if;
    delete from alertes_whatsapp where avis_id = a.id and type = 'avis_reponse';
    insert into alertes_whatsapp (commerce_id, avis_id, type, canal, prioritaire, envoyer_apres, donnees)
      values (a.commerce_id, a.id, 'avis_reponse', 'whatsapp', true, ap,
              jsonb_build_object('note', a.note, 'auteur', nullif(auteur_reduit(a.auteur), ''), 'reponse', left(t, 4096), 'texte', left(a.texte, 4096),
                                 'pas_avant', pa, 'ecrit_le', a.ecrit_le, 'sensible', coalesce(a.sensible, false) or a.statut = 'humain',
                                 'echeance', a.cree_le + interval '24 hours'));
    -- Contre-revue (E) : seul un NOUVEL envoi lève l'attente (le message du commerçant est pris en compte, le délai repart) et ferme « à renvoyer ».
    update avis set retenue_le = null, attends_le = null where id = a.id and retenue_le is not null;
    update avis set suspendu_le = null, suspendu_motif = null where id = a.id and suspendu_le is not null;   -- contre-revue n°2
    update alertes_exploitant set vue_le = now() where vue_le is null and detail->>'avis' = a.id::text and type in ('AVIS_A_RENVOYER','AVIS_RETENU');
  end if;
  insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_PROPOSE', jsonb_build_object('avis', a.id, 'sensible', a.sensible, 'whatsapp', wa, 'pas_avant', pa));
  return jsonb_build_object('id', a.id, 'propose_le', now(), 'email_client', c.email_validation, 'whatsapp', wa,
                            'sensible', coalesce(a.sensible, false) or a.statut = 'humain', 'pas_avant', pa, 'envoi_le', ap);
end $$;
revoke all on function admin_avis_proposer(uuid, text) from public, anon;
grant execute on function admin_avis_proposer(uuid, text) to authenticated;   -- vérifie est_admin()

-- 4. Départ réel de l'alerte : whatsapp_a_envoyer réserve le créneau (statut « en_cours ») ; l'heure annoncée est calculée à ce moment, écrite
-- dans l'alerte (le message la lit) et sur l'avis (rien ne part avant). Avis de 1 à 3 étoiles avec une réponse seulement.
create or replace function annoncer_heure_publication() returns trigger language plpgsql security definer set search_path = public as $$
declare a avis; pa timestamptz;
begin
  if new.avis_id is null or coalesce((new.donnees->>'sensible')::boolean, false) or nullif(btrim(coalesce(new.donnees->>'reponse', '')), '') is null then return new; end if;
  -- Contre-revue (A) : réponse renvoyée après un ATTENDS : aucune heure (publication 48 h après l'ATTENDS, sauf nouveau message).
  if coalesce((new.donnees->>'en_attente')::boolean, false) then return new; end if;
  select * into a from avis where id = new.avis_id;
  if a.id is null or a.statut <> 'escalade' or coalesce(a.sensible, false) then return new; end if;
  pa := avis_heure_annoncee(greatest(now(), coalesce(new.creneau, now())), a.cree_le);
  new.donnees := new.donnees || jsonb_build_object('pas_avant', pa);
  update avis set publication_pas_avant = pa where id = a.id;
  return new;
end $$;
revoke all on function annoncer_heure_publication() from public, anon, authenticated;
drop trigger if exists annoncer_heure_publication on alertes_whatsapp;
create trigger annoncer_heure_publication before update of statut on alertes_whatsapp for each row
  when (new.statut = 'en_cours' and old.statut = 'a_envoyer' and new.canal_envoi = 'whatsapp' and new.type in ('avis_recu','avis_reponse'))
  execute function annoncer_heure_publication();
-- whatsapp_a_envoyer (version du lot 10) rend l'heure lue sur la ligne APRÈS la réservation (remplacement d'un seul motif, rejouable).
do $l11$
declare d text := pg_get_functiondef('whatsapp_a_envoyer()'::regprocedure); m text := '''pas_avant'', x.donnees->>''pas_avant''';
  m2 text := '''avis_id'', coalesce(x.avis_id::text, x.donnees->>''avis''),';
begin
  if position(m in d) > 0 then
    d := replace(d, m, '''pas_avant'', (select w.donnees->>''pas_avant'' from alertes_whatsapp w where w.id = x.id)');
  end if;
  -- Revue : la référence courte de l'avis (« Réf. A7K ») et « publiée automatiquement » (avis_traite) dans chaque alerte.
  if position('ref_courte' in d) = 0 and position(m2 in d) > 0 then
    d := replace(d, m2, m2 || ' ''ref'', (select v.ref_courte from avis v where v.id = x.avis_id), ''auto'', coalesce((x.donnees->>''auto'')::boolean, false),');
  end if;
  -- Contre-revue (A) : « en_attente » (réponse renvoyée après un ATTENDS, sans heure de publication).
  if position('en_attente' in d) = 0 and position('''auto'', coalesce((x.donnees->>''auto'')::boolean, false),' in d) > 0 then
    d := replace(d, '''auto'', coalesce((x.donnees->>''auto'')::boolean, false),', '''auto'', coalesce((x.donnees->>''auto'')::boolean, false), ''en_attente'', coalesce((x.donnees->>''en_attente'')::boolean, false),');
  end if;
  if d is distinct from pg_get_functiondef('whatsapp_a_envoyer()'::regprocedure) then execute d; end if;
end $l11$;

-- 5. Délivrance (resultat_whatsapp, version du lot 8 avec les types du lot 10, recopiée) : la règle des 3 h du lot 8 (qui déplaçait l'heure après
-- l'envoi du message, donc différente de celle écrite dedans) est retirée ; l'heure annoncée est posée par noter_reponse_annoncee.
create or replace function resultat_whatsapp(p_id uuid, p_canal text, p_ok boolean, p_erreur text default null, p_ref text default null) returns jsonb
language plpgsql security definer set search_path = public as $$
declare a alertes_whatsapp; d jsonb;
begin
  select * into a from alertes_whatsapp where id = p_id for update;
  if a.id is null then raise exception 'Alerte inconnue.' using errcode = 'P0080'; end if;
  if p_canal is null or p_canal not in ('whatsapp','email','aucun') then raise exception 'Canal inconnu : %', p_canal using errcode = 'P0080'; end if;
  if a.statut <> 'en_cours' or (p_canal = 'whatsapp' and a.canal_envoi is distinct from 'whatsapp') then
    return jsonb_build_object('id', a.id, 'statut', a.statut, 'deja', true);
  end if;
  d := jsonb_build_object('alerte', a.id, 'type', a.type, 'avis', a.avis_id);
  if p_canal = 'whatsapp' and coalesce(p_ok, false) then
    update alertes_whatsapp set statut = 'envoye', traite_le = now(), ref = left(p_ref, 100), erreur = null, donnees = '{}'::jsonb where id = a.id;
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'WHATSAPP_ENVOYE', d || jsonb_build_object('ref', left(p_ref, 100)));
  elsif p_canal = 'whatsapp' and a.type in ('avis_sensible','avis_recu','avis_rappel_message','avis_reponse') then   -- alerte garantie : l'équipe est alertée
    update alertes_whatsapp set statut = 'echec', traite_le = now(), erreur = left(coalesce(p_erreur, 'erreur inconnue'), 500) where id = a.id;
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'ALERTE_WHATSAPP_ECHEC', d || jsonb_build_object('erreur', left(coalesce(p_erreur, 'erreur inconnue'), 500)));
    insert into alertes_exploitant (type, commerce_id, detail) values ('ALERTE_WHATSAPP_ECHEC', a.commerce_id, jsonb_build_object('avis', a.avis_id, 'alerte', a.id, 'erreur', left(coalesce(p_erreur, 'erreur inconnue'), 500)));
  elsif p_canal = 'whatsapp' then
    update alertes_whatsapp set canal_envoi = 'email', raison_email = 'echec_whatsapp', erreur = left(coalesce(p_erreur, 'erreur inconnue'), 500) where id = a.id;
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'WHATSAPP_ECHEC', d || jsonb_build_object('erreur', left(coalesce(p_erreur, 'erreur inconnue'), 500)));
  elsif p_canal = 'email' and coalesce(p_ok, false) then
    update alertes_whatsapp set statut = 'email', traite_le = now(), donnees = '{}'::jsonb, raison_email = coalesce(raison_email, left(nullif(p_erreur, ''), 100))
      where id = a.id returning * into a;
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'ALERTE_PAR_EMAIL', d || jsonb_build_object('raison', a.raison_email));
  elsif p_canal = 'email' then
    update alertes_whatsapp set statut = 'echec', traite_le = now(), erreur = left(coalesce(p_erreur, 'erreur inconnue'), 500), donnees = '{}'::jsonb where id = a.id;
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'ALERTE_EMAIL_ECHEC', d || jsonb_build_object('erreur', left(coalesce(p_erreur, 'erreur inconnue'), 500)));
  else
    update alertes_whatsapp set statut = 'abandonne', traite_le = now(), donnees = '{}'::jsonb where id = a.id;
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'ALERTE_SANS_CANAL', d || jsonb_build_object('raison', coalesce(a.raison_email, p_erreur)));
  end if;
  select * into a from alertes_whatsapp where id = p_id;
  return jsonb_build_object('id', a.id, 'statut', a.statut, 'canal_envoi', a.canal_envoi);
end $$;
revoke all on function resultat_whatsapp(uuid, text, boolean, text, text) from public, anon, authenticated;
grant execute on function resultat_whatsapp(uuid, text, boolean, text, text) to service_role;

-- Texte reçu par le commerçant (lot 10) : seulement avis_recu et avis_reponse (l'alerte d'arrivée d'un avis sensible ne porte plus de réponse).
-- 1 à 3 étoiles : publication à l'heure écrite dans le message, jamais moins de 4 h après la délivrance (alerte réservée avant ce bloc, ou
-- délivrée en retard : l'heure ne recule jamais, elle peut seulement venir plus tard que celle écrite).
create or replace function noter_reponse_annoncee() returns trigger language plpgsql security definer set search_path = public as $$
declare x text := nullif(btrim(coalesce(old.donnees->>'reponse', '')), ''); pa timestamptz;
begin
  if new.avis_id is null or x is null or new.type not in ('avis_recu','avis_reponse') then return null; end if;
  update avis set reponse_annoncee = x, reponse_annoncee_le = now() where id = new.avis_id and statut in ('escalade','humain');
  if not coalesce((old.donnees->>'sensible')::boolean, false) and not coalesce((old.donnees->>'en_attente')::boolean, false) then
    pa := nullif(old.donnees->>'pas_avant', '')::timestamptz;
    update avis set publication_pas_avant = greatest(coalesce(pa, now()), now() + make_interval(hours => (avis_delais_annonce()->>'min_heures')::int) - interval '5 minutes')
     where id = new.avis_id and statut = 'escalade';
  end if;
  return null;
end $$;
revoke all on function noter_reponse_annoncee() from public, anon, authenticated;
drop trigger if exists reponse_annoncee on alertes_whatsapp;
create trigger reponse_annoncee after update of statut on alertes_whatsapp for each row
  when (new.statut = 'envoye' and old.statut is distinct from 'envoye' and new.type in ('avis_recu','avis_reponse'))
  execute function noter_reponse_annoncee();

-- ---------------------------------------------------------------- 6. Publication par l'API Google, sans clic de l'équipe
-- Règles (une seule fonction, lue par la base et par n8n) : rend « ok » (OK écrit du commerçant reçu APRÈS la réponse qu'il a reçue), ou
-- « heure_annoncee » (avis de 1 à 3 étoiles, heure annoncée passée, sans message), sinon null. Toujours : avis reçu par l'API (pas Places),
-- commerce relié et actif, aucune demande d'attendre, réponse reçue par WhatsApp (alerte délivrée) et texte inchangé depuis, aucune nouvelle
-- réponse en train de partir, pas déjà prise par n8n depuis moins de 30 minutes. Avis sensible : « ok » seulement (jamais sans OK écrit).
create or replace function avis_publiable_api(p_id uuid) returns text language plpgsql stable security definer set search_path = public as $$
declare a avis; c commerces; o offres;
begin
  select * into a from avis where id = p_id;
  if a.id is null or a.statut not in ('escalade','humain') or a.retenue_le is not null then return null; end if;
  if a.faux_signale_le is not null or a.renvoi_attendu_le is not null then return null; end if;   -- revue : FAUX bloque pour toujours ; réponse corrigée à renvoyer
  -- Contre-revues : dans le doute, on ne publie pas. Avis suspendu (message pas compris ou pas rattaché) ; avis débloqué après FAUX tant que
  -- l'équipe n'a pas renvoyé de nouvelle réponse. Avant le OK : un avis accepté mais suspendu n'est pas publié.
  if a.suspendu_le is not null then return null; end if;
  if exists (select 1 from alertes_exploitant e where e.vue_le is null and e.type = 'AVIS_A_RENVOYER' and e.detail->>'avis' = a.id::text) then return null; end if;
  if coalesce(a.source, 'api') = 'places' or nullif(btrim(coalesce(a.google_review_id, '')), '') is null then return null; end if;
  if a.reponse_annoncee_le is null or nullif(btrim(coalesce(a.reponse_annoncee, '')), '') is null
     or btrim(coalesce(a.reponse_proposee, '')) is distinct from btrim(a.reponse_annoncee) then return null; end if;
  if a.api_prise_le is not null and a.api_prise_le > now() - interval '30 minutes' then return null; end if;
  if exists (select 1 from alertes_whatsapp w where w.avis_id = a.id and w.type = 'avis_reponse' and w.statut in ('a_envoyer','en_cours')) then return null; end if;
  select * into c from commerces where id = a.commerce_id;
  if c.id is null or c.en_pause or nullif(btrim(coalesce(c.google_location_id, '')), '') is null then return null; end if;
  if a.decision_client = 'approuve' and a.decision_client_le >= a.reponse_annoncee_le then return 'ok'; end if;
  if coalesce(a.sensible, false) or a.statut = 'humain' or coalesce(a.note, 5) > 3 then return null; end if;
  if a.version_client_le is not null and a.version_client_le >= a.reponse_annoncee_le then return null; end if;   -- message après la réponse reçue
  select * into o from offres where id = c.offre_id;
  if not coalesce(o.avis_equipe or o.delegation, false) then return null; end if;
  if a.publication_pas_avant is null or a.publication_pas_avant > now() then return null; end if;
  return 'heure_annoncee';
end $$;
revoke all on function avis_publiable_api(uuid) from public, anon, authenticated;

-- Demande de publication : la base pose api_demande_le (le déclencheur hook_avis_publier appelle n8n). n8n prend l'avis (avis_a_publier_api :
-- règles relues, verrou), publie, rend le résultat (avis_resultat_api).
create or replace function demander_publication_api(p_id uuid, p_motif text) returns boolean language plpgsql security definer set search_path = public as $$
begin
  update avis set api_demande_le = now(), api_motif = p_motif where id = p_id;
  insert into journal (commerce_id, type, detail) select commerce_id, 'AVIS_PUBLICATION_API_DEMANDEE', jsonb_build_object('avis', id, 'motif', p_motif) from avis where id = p_id;
  return found;
end $$;
revoke all on function demander_publication_api(uuid, text) from public, anon, authenticated;

create or replace function avis_a_publier_api(p_avis uuid) returns jsonb language plpgsql security definer set search_path = public as $$
declare a avis; c commerces; m text;
begin
  select * into a from avis where id = p_avis for update;
  if a.id is null then return jsonb_build_object('publier', false, 'raison', 'avis inconnu'); end if;
  m := avis_publiable_api(a.id);
  if m is null then return jsonb_build_object('publier', false, 'raison', 'plus publiable (attente, nouveau message, texte changé ou déjà publié)', 'avis', a.id); end if;
  select * into c from commerces where id = a.commerce_id;
  update avis set api_prise_le = now(), api_motif = case when a.api_motif = 'attente_48h' then a.api_motif else m end where id = a.id;   -- motif 48 h gardé
  return jsonb_build_object('publier', true, 'avis', a.id, 'commerce_id', a.commerce_id, 'motif', case when a.api_motif = 'attente_48h' then a.api_motif else m end, 'texte', btrim(a.reponse_annoncee),
                            'location', c.google_location_id, 'review', a.google_review_id);
end $$;
revoke all on function avis_a_publier_api(uuid) from public, anon, authenticated;
grant execute on function avis_a_publier_api(uuid) to service_role;   -- n8n seulement

create or replace function avis_resultat_api(p_avis uuid, p_ok boolean, p_texte text default null, p_erreur text default null) returns jsonb language plpgsql security definer set search_path = public as $$
declare a avis; t text := nullif(btrim(coalesce(p_texte, '')), ''); ech timestamptz;
begin
  select * into a from avis where id = p_avis for update;
  if a.id is null or a.statut not in ('escalade','humain') then return jsonb_build_object('etat', 'deja', 'statut', a.statut); end if;
  ech := echeance_avis(a.id, a.cree_le);
  if coalesce(p_ok, false) and t is not null then
    update avis set reponse = t, statut = 'publie', publie_par = 'api', erreur = null, api_prise_le = null where id = a.id;
    update alertes_exploitant set vue_le = now() where vue_le is null and detail->>'avis' = a.id::text
       and type in ('AVIS_ACCORD_A_PUBLIER','AVIS_RETENU','AVIS_VERSION_CLIENT','AVIS_CORRECTION_A_RELIRE');
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_PUBLIE', jsonb_build_object('avis', a.id, 'par', 'api', 'motif', a.api_motif,
      'accord_client', a.api_motif = 'ok', 'recu_le', a.cree_le, 'echeance', ech, 'dans_les_24h', now() <= coalesce(ech, 'infinity'::timestamptz)));
    return jsonb_build_object('etat', 'publie', 'avis', a.id);
  end if;
  update avis set statut = 'a_publier_main', reponse_proposee = coalesce(t, reponse_proposee), api_prise_le = null,
    erreur = left('Publication par l''API refusée : ' || coalesce(nullif(btrim(p_erreur), ''), 'erreur inconnue'), 300) where id = a.id;
  insert into journal (commerce_id, type, detail) values (a.commerce_id, 'SECOURS_A_PUBLIER', jsonb_build_object('avis', a.id, 'motif', a.api_motif, 'erreur', left(coalesce(p_erreur, ''), 300)));
  return jsonb_build_object('etat', 'a_publier_main', 'avis', a.id);
end $$;
revoke all on function avis_resultat_api(uuid, boolean, text, text) from public, anon, authenticated;
grant execute on function avis_resultat_api(uuid, boolean, text, text) to service_role;   -- n8n seulement

-- ---------------------------------------------------------------- 7. Réponses du commerçant (corps du lot 8, recopié ; revue du 17/11/2026)
-- OK : publication immédiate par l'API quand c'est possible (sinon tâche AVIS_ACCORD_A_PUBLIER, comme avant : avis Places, réponse changée…) ;
-- avis sensible : OK pris seulement sur une réponse ENVOYÉE par l'équipe (propose_le). ATTENDS : publication 48 h après, sauf nouveau message.
-- FAUX : faux_signale_le, rien n'est plus jamais publié jusqu'à « Débloquer » (équipe) ; tout message reçu ensuite sur cet avis est une
-- explication pour l'équipe, ajoutée à la tâche FAUX (pas de réécriture, rien n'est renvoyé, l'attente reste). Correction : ajoutée à
-- avis.corrections (toutes gardées, la dernière gagne), Claude réécrit avec toutes (n8n) ; une réponse corrigée plus ancienne encore en file
-- est retirée (jamais de version périmée envoyée).
-- Résumé d'un avis pour l'accusé de réception et la liste « lequel ? » : référence, prénom et initiale, début de l'avis.
create or replace function avis_resume(a avis) returns jsonb language sql stable set search_path = public as $$
  select jsonb_build_object('ref', a.ref_courte, 'auteur', nullif(auteur_reduit(a.auteur), ''), 'note', a.note, 'sensible', coalesce(a.sensible, false),
    'debut', case when char_length(coalesce(a.texte, '')) > 60 then left(regexp_replace(a.texte, '\s+', ' ', 'g'), 57) || '…' else coalesce(a.texte, '') end) $$;
revoke all on function avis_resume(avis) from public, anon, authenticated;
-- Avis « en attente du commerçant » : en file, une alerte lui a été délivrée (arrivée, réponse proposée), pas déjà approuvé, pas signalé FAUX
-- depuis plus de 48 h (au-delà, seul un message qui reprend sa référence le vise).
create or replace function avis_attend_commercant(a avis) returns boolean language sql stable set search_path = public as $$
  select a.statut in ('escalade','humain') and (coalesce(a.sensible, false) or coalesce(a.note, 5) <= 3)
     and exists (select 1 from alertes_whatsapp w where w.avis_id = a.id and w.type in ('avis_recu','avis_sensible','avis_reponse') and w.statut = 'envoye')
     and not coalesce(a.decision_client = 'approuve' and a.decision_client_le >= coalesce(a.reponse_annoncee_le, a.propose_le, a.cree_le), false)
     and (a.faux_signale_le is null or a.faux_signale_le > now() - interval '48 hours') $$;
revoke all on function avis_attend_commercant(avis) from public, anon, authenticated;

-- Contre-revue n°2 (point 4) : avis « en cours » pour le rattachement : en attente du commerçant, ou accepté (OK) mais pas encore publié.
create or replace function avis_en_cours_commercant(a avis) returns boolean language sql stable set search_path = public as $$
  select a.statut in ('escalade','humain') and (coalesce(a.sensible, false) or coalesce(a.note, 5) <= 3)
     and exists (select 1 from alertes_whatsapp w where w.avis_id = a.id and w.type in ('avis_recu','avis_sensible','avis_reponse') and w.statut = 'envoye')
     and (a.faux_signale_le is null or a.faux_signale_le > now() - interval '48 hours') $$;
revoke all on function avis_en_cours_commercant(avis) from public, anon, authenticated;

create or replace function avis_message_commercant(p_avis uuid, p_choix text, p_texte text, p_canal text) returns jsonb language plpgsql security definer set search_path = public as $$
declare a avis; t timestamptz := now(); x text := left(nullif(btrim(coalesce(p_texte, '')), ''), 2000); ch text := lower(coalesce(p_choix, '')); m text; e record;
begin
  select * into a from avis where id = p_avis for update;
  if a.id is null or a.statut not in ('escalade','humain') then return jsonb_build_object('etat', 'plus_en_attente', 'avis', p_avis); end if;
  if ch not in ('ok','attends','faux','correction') then raise exception 'Choix inconnu.' using errcode = 'P0090'; end if;
  -- Revue, point 2 : après FAUX, tout autre message est une explication pour l'équipe, ajoutée à la tâche FAUX.
  if a.faux_signale_le is not null and ch <> 'faux' then
    select * into e from alertes_exploitant where type = 'AVIS_FAUX_SIGNALE' and detail->>'avis' = a.id::text and vue_le is null order by cree_le desc limit 1;
    if e.id is null then
      insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_FAUX_SIGNALE', a.commerce_id,
        jsonb_build_object('avis', a.id, 'note', a.note, 'canal', p_canal, 'urgent', true, 'messages', jsonb_build_array(jsonb_build_object('le', t, 'texte', coalesce(x, ch)))));
    else
      update alertes_exploitant set detail = detail || jsonb_build_object('messages', coalesce(detail->'messages', '[]'::jsonb) || jsonb_build_array(jsonb_build_object('le', t, 'texte', coalesce(x, ch))))
       where id = e.id;
    end if;
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_FAUX_EXPLICATION', jsonb_build_object('avis', a.id, 'canal', p_canal));
    return jsonb_build_object('etat', 'faux_explication', 'choix', ch, 'avis', a.id, 'commerce_id', a.commerce_id, 'publication_api', false) || avis_resume(a);
  end if;
  if ch = 'ok' and (nullif(btrim(coalesce(a.reponse_proposee, '')), '') is null
                    or ((coalesce(a.sensible, false) or a.statut = 'humain') and a.propose_le is null)) then ch := 'attends'; end if;   -- rien à approuver encore
  if ch = 'correction' and x is null then raise exception 'Écrivez votre message en quelques mots.' using errcode = 'P0090'; end if;
  if ch = 'ok' then
    update avis set decision_client = 'approuve', decision_client_le = t, retenue_le = null, attends_le = null where id = a.id;
    update alertes_exploitant set vue_le = t where type = 'AVIS_RETENU' and commerce_id = a.commerce_id and detail->>'avis' = a.id::text and vue_le is null;
    m := avis_publiable_api(a.id);
    if m = 'ok' then
      perform demander_publication_api(a.id, 'ok');
    elsif a.suspendu_le is null then   -- avis suspendu : rien à publier à la main ; « Appliquer OK » de l'équipe le publiera
      insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_ACCORD_A_PUBLIER', a.commerce_id, jsonb_build_object('avis', a.id, 'note', a.note, 'canal', p_canal, 'urgent', true));
    end if;
  elsif ch = 'attends' then
    -- Contre-revue (A) : un ATTENDS est toujours une vraie attente, datée : les 48 h repartent de lui (même après une correction, même déjà en attente) ;
    -- une réponse corrigée pas encore partie part sans heure de publication.
    update avis set retenue_le = t, attends_le = t, api_demande_le = null, decision_client = case when decision_client = 'approuve' then null else decision_client end where id = a.id;
    update alertes_exploitant set vue_le = t where type = 'AVIS_ACCORD_A_PUBLIER' and detail->>'avis' = a.id::text and vue_le is null;   -- l'accord est retiré
    update alertes_whatsapp set donnees = (donnees - 'pas_avant') || jsonb_build_object('en_attente', true)
     where avis_id = a.id and type = 'avis_reponse' and statut = 'a_envoyer';
    insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_RETENU', a.commerce_id, jsonb_build_object('avis', a.id, 'note', a.note, 'mode', 'message', 'canal', p_canal));
  elsif ch = 'faux' then
    update avis set faux_signale_le = coalesce(faux_signale_le, t), retenue_le = coalesce(retenue_le, t), api_demande_le = null, renvoi_attendu_le = null,
                    decision_client = case when decision_client = 'approuve' then null else decision_client end where id = a.id;
    update alertes_exploitant set vue_le = t where type = 'AVIS_ACCORD_A_PUBLIER' and detail->>'avis' = a.id::text and vue_le is null;   -- l'accord est retiré
    delete from alertes_whatsapp where avis_id = a.id and type = 'avis_reponse' and statut = 'a_envoyer';
    if not exists (select 1 from alertes_exploitant where type = 'AVIS_FAUX_SIGNALE' and detail->>'avis' = a.id::text and vue_le is null) then
      insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_FAUX_SIGNALE', a.commerce_id, jsonb_build_object('avis', a.id, 'note', a.note, 'canal', p_canal, 'urgent', true));
    end if;
  else
    -- Point 4 : toutes les corrections gardées (la dernière gagne) ; une réponse corrigée plus ancienne pas encore partie est retirée.
    update avis set corrections = coalesce(corrections, '[]'::jsonb) || jsonb_build_array(jsonb_build_object('le', t, 'texte', x)),
                    version_client = x, version_client_le = t, retenue_le = coalesce(retenue_le, t), api_demande_le = null, renvoi_attendu_le = null where id = a.id;
    delete from alertes_whatsapp where avis_id = a.id and type = 'avis_reponse' and statut = 'a_envoyer';
  end if;
  insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_MESSAGE_COMMERCANT', jsonb_build_object('avis', a.id, 'choix', ch, 'canal', p_canal, 'api', m));
  return jsonb_build_object('etat', 'ok', 'choix', ch, 'avis', a.id, 'commerce_id', a.commerce_id, 'publication_api', coalesce(m = 'ok', false)) || avis_resume(a);
end $$;
revoke all on function avis_message_commercant(uuid, text, text, text) from public, anon, authenticated;

-- Point 3 : rattachement SÛR d'un message WhatsApp à son avis (revue, puis contre-revue du 17/11/2026).
--   1. des références (« OK A7K ») : TOUTES doivent être celles d'un même avis en file des commerces de ce numéro ; sinon (référence inconnue,
--      avis déjà traité, plusieurs avis, même référence dans deux commerces) : « lequel ? », JAMAIS de repli sur l'avis en attente ;
--   2. sans référence : l'unique avis en attente du commerçant (avis_attend_commercant), avis sensibles compris ; zéro ou plusieurs, ou numéro
--      partagé par plusieurs commerces : « lequel ? ».
--   « Lequel ? » : RIEN n'est appliqué, n8n répond avec la liste (référence, auteur, début de l'avis), tâche WHATSAPP_A_RATTACHER (avec le message),
--   les avis en attente sont mis en attente (délai de 24 h suspendu comme pour ATTENDS, CGV art. 9 et 10) et aucun avis de ces commerces n'est
--   publié tant que la tâche est ouverte (message_commercant_en_suspens).
--   Mot-clé avec un autre texte (« FAUX, je ne connais pas… ») : tâche WHATSAPP_A_LIRE, avis mis en attente, rien n'est appliqué. Message sans
--   texte (photo, vocal) : tâche WHATSAPP_RECU, publication suspendue.
-- L'accusé de réception dit toujours l'avis compris (ref, auteur, debut rendus). Le message cité par WhatsApp (réponse à un message) n'est pas lu :
-- la documentation de WaSender consultée ne le décrit pas (ordres/LOT11.md).
create or replace function whatsapp_a_preciser(p_cs uuid[], p_texte text, p_ref text, p_motif text, p_avis uuid default null) returns jsonb language plpgsql security definer set search_path = public as $$
declare c commerces; liste jsonb; ids uuid[];
begin
  select * into c from commerces where id = p_cs[1];
  select coalesce(jsonb_agg(avis_resume(v) order by v.cree_le), '[]'::jsonb), array_agg(v.id) into liste, ids from avis v
   where v.commerce_id = any(p_cs) and (avis_en_cours_commercant(v) or v.id = p_avis);
  if jsonb_array_length(liste) = 0 and p_motif = 'sans_reference' then   -- aucun avis en cours : information seulement, rien n'est suspendu
    insert into journal (commerce_id, type, detail) values (c.id, 'WHATSAPP_RECU_SANS_AVIS', jsonb_build_object('ref', p_ref));
    insert into alertes_exploitant (type, commerce_id, detail) values ('WHATSAPP_RECU', c.id, jsonb_build_object('recu_le', now(), 'commerce_ids', to_jsonb(p_cs)));
    return jsonb_build_object('etat', 'sans_avis', 'commerce', c.nom, 'langue', coalesce(c.langue, 'fr'));
  end if;
  -- Seuls les avis de la liste (en cours AU MOMENT du message) sont suspendus ; leur délai de 24 h l'est aussi (avis_suspensions, CGV art. 9).
  perform suspendre_avis(ids, 'a_rattacher');
  insert into journal (commerce_id, type, detail) values (c.id, 'WHATSAPP_RECU_A_PRECISER', jsonb_build_object('ref', p_ref, 'motif', p_motif, 'en_attente', jsonb_array_length(liste), 'avis', p_avis));
  insert into alertes_exploitant (type, commerce_id, detail) values ('WHATSAPP_A_RATTACHER', c.id,
    jsonb_build_object('recu_le', now(), 'texte', left(p_texte, 500), 'avis', liste, 'avis_ids', to_jsonb(coalesce(ids, '{}')), 'commerces', cardinality(p_cs),
                       'commerce_ids', to_jsonb(p_cs), 'motif', p_motif, 'urgent', true));
  return jsonb_build_object('etat', 'a_preciser', 'motif', p_motif, 'liste', liste, 'commerce', c.nom, 'langue', coalesce(c.langue, 'fr'));
end $$;
revoke all on function whatsapp_a_preciser(uuid[], text, text, text, uuid) from public, anon, authenticated;

create or replace function whatsapp_recu(p_numero text, p_texte text, p_ref text default null) returns jsonb language plpgsql security definer set search_path = public as $$
declare n text := '+' || regexp_replace(coalesce(p_numero, ''), '\D', '', 'g'); cs uuid[]; c commerces; a avis; refs text[] := refs_du_message(p_texte);
  ids uuid[]; connues text[]; reste text := coalesce(p_texte, ''); ch text; r jsonb; k text; motif text; liste jsonb;
begin
  if p_ref is not null and exists (select 1 from journal where type in ('WHATSAPP_RECU','WHATSAPP_RECU_SANS_AVIS','WHATSAPP_RECU_A_PRECISER','WHATSAPP_RECU_A_LIRE') and detail->>'ref' = p_ref) then
    return jsonb_build_object('etat', 'deja'); end if;
  -- 16/11/2026 : un numéro qui n'est pas celui d'un commerce ACTIF est ignoré SANS AUCUNE écriture (WhatsApp personnel du fondateur).
  if n !~ '^\+[1-9][0-9]{7,14}$' then return jsonb_build_object('etat', 'inconnu'); end if;
  select array_agg(id order by cree_le) into cs from commerces where whatsapp_numero = n and (fin_contrat_le is null or fin_contrat_le >= current_date);
  if cs is null then return jsonb_build_object('etat', 'inconnu'); end if;
  select * into c from commerces where id = cs[1];
  -- Message sans texte (photo, vocal, document) : les avis en cours sont suspendus (l'équipe le lit) ; aucun avis en cours : information seulement.
  if lire_message_whatsapp(p_texte) is null then
    select coalesce(jsonb_agg(avis_resume(v) order by v.cree_le), '[]'::jsonb), array_agg(v.id) into liste, ids from avis v where v.commerce_id = any(cs) and avis_en_cours_commercant(v);
    insert into journal (commerce_id, type, detail) values (c.id, 'WHATSAPP_RECU_A_LIRE', jsonb_build_object('ref', p_ref, 'vide', true, 'en_attente', jsonb_array_length(liste)));
    if ids is null then
      insert into alertes_exploitant (type, commerce_id, detail) values ('WHATSAPP_RECU', c.id, jsonb_build_object('recu_le', now(), 'vide', true, 'commerce_ids', to_jsonb(cs)));
    else
      perform suspendre_avis(ids, 'sans_texte');
      insert into alertes_exploitant (type, commerce_id, detail) values ('WHATSAPP_A_LIRE', c.id,
        jsonb_build_object('recu_le', now(), 'vide', true, 'avis', liste, 'avis_ids', to_jsonb(ids), 'commerce_ids', to_jsonb(cs), 'urgent', true));
    end if;
    return jsonb_build_object('etat', 'vide', 'commerce', c.nom, 'langue', coalesce(c.langue, 'fr'), 'numero', n);
  end if;
  if cardinality(refs) > 0 then
    select array_agg(distinct v.id), array_agg(distinct upper(v.ref_courte)) into ids, connues from avis v
     where v.commerce_id = any(cs) and v.statut in ('escalade','humain') and upper(v.ref_courte) = any(refs);
    if cardinality(ids) = 1 and cardinality(connues) = cardinality(refs) then
      select * into a from avis where id = ids[1];
    else
      motif := case when coalesce(cardinality(ids), 0) = 0 then 'reference_inconnue' when cardinality(ids) > 1 then 'plusieurs_avis' else 'reference_inconnue' end;
    end if;
  elsif cardinality(cs) = 1 then
    select array_agg(v.id) into ids from avis v where v.commerce_id = c.id and avis_en_cours_commercant(v);
    if cardinality(ids) = 1 then select * into a from avis where id = ids[1]; else motif := 'sans_reference'; end if;
  else
    motif := 'numero_partage';
  end if;
  if a.id is null then return whatsapp_a_preciser(cs, p_texte, p_ref, motif) || jsonb_build_object('numero', n); end if;
  foreach k in array refs loop reste := regexp_replace(reste, '\m' || k || '\M', ' ', 'gi'); end loop;
  ch := lire_message_whatsapp(reste);
  reste := btrim(regexp_replace(reste, '^[[:space:][:punct:]]*(r[ée]f[ée]rence|r[ée]f)?[[:space:][:punct:]]*', '', 'i'));
  if ch is null then   -- la référence seule : on ne devine rien, on redemande pour cet avis
    return whatsapp_a_preciser(cs, p_texte, p_ref, 'reference_seule', a.id) || jsonb_build_object('numero', n);
  end if;
  select * into c from commerces where id = a.commerce_id;
  if ch = 'a_lire' and a.faux_signale_le is not null then ch := 'correction'; end if;   -- après FAUX : explication pour l'équipe (avis_message_commercant)
  if ch = 'a_lire' then
    perform suspendre_avis(array[a.id], 'a_lire');
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'WHATSAPP_RECU_A_LIRE', jsonb_build_object('ref', p_ref, 'avis', a.id));
    insert into alertes_exploitant (type, commerce_id, detail) values ('WHATSAPP_A_LIRE', a.commerce_id,
      jsonb_build_object('avis', a.id, 'avis_ids', to_jsonb(array[a.id]), 'note', a.note, 'ref', a.ref_courte, 'recu_le', now(), 'texte', left(p_texte, 500),
                         'commerce_ids', to_jsonb(array[a.commerce_id]), 'urgent', true));
    return jsonb_build_object('etat', 'a_lire', 'avis', a.id, 'commerce_id', a.commerce_id, 'commerce', c.nom, 'langue', coalesce(c.langue, 'fr'), 'numero', n,
                              'sensible', coalesce(a.sensible, false)) || avis_resume(a);
  end if;
  r := avis_message_commercant(a.id, ch, reste, 'whatsapp');
  insert into journal (commerce_id, type, detail) values (a.commerce_id, 'WHATSAPP_RECU', jsonb_build_object('ref', p_ref, 'avis', a.id, 'choix', ch, 'par_reference', cardinality(refs) > 0));
  return r || jsonb_build_object('commerce', c.nom, 'langue', coalesce(c.langue, 'fr'), 'numero', n, 'sensible', coalesce(a.sensible, false));
end $$;
revoke all on function whatsapp_recu(text, text, text) from public, anon, authenticated;
grant execute on function whatsapp_recu(text, text, text) to service_role;   -- n8n seulement

-- Contre-revue n°2 : « Appliquer OK », « Appliquer ATTENDS », « Appliquer FAUX » (console, sur un avis suspendu ou en attente) : l'équipe a lu le
-- message du commerçant et applique sa réponse en son nom. Même logique que sa réponse (avis_message_commercant, canal « equipe »), geste
-- journalisé (AVIS_REPONSE_APPLIQUEE : administrateur, heure, suspension levée). Lève la suspension ; ATTENDS ou FAUX retirent un accord
-- (tâche AVIS_ACCORD_A_PUBLIER fermée). Avis signalé FAUX : « Débloquer » d'abord.
create or replace function admin_appliquer_reponse(p_avis uuid, p_choix text) returns jsonb language plpgsql security definer set search_path = public as $$
declare a avis; ch text := lower(coalesce(p_choix, '')); r jsonb;
begin
  if not est_admin() then raise exception 'réservé à l''administrateur' using errcode = 'P0005'; end if;
  if ch not in ('ok','attends','faux') then raise exception 'Choix inconnu.' using errcode = 'P0090'; end if;
  select * into a from avis where id = p_avis for update;
  if a.id is null or a.statut not in ('escalade','humain') then raise exception 'Cet avis n''est pas dans la file de BoostImpact.' using errcode = 'P0061'; end if;
  if a.faux_signale_le is not null then raise exception 'Avis signalé FAUX : « Débloquer » d''abord.' using errcode = 'P0061'; end if;
  -- Contre-revue n°3 : OK seulement sur le texte que le commerçant a reçu (alerte délivrée, texte identique) : c'est lui qui part.
  if ch = 'ok' and (a.reponse_annoncee_le is null or nullif(btrim(coalesce(a.reponse_annoncee, '')), '') is null
                    or btrim(coalesce(a.reponse_proposee, '')) is distinct from btrim(a.reponse_annoncee)) then
    raise exception 'Le commerçant n''a jamais reçu ce texte : « Envoyer au commerçant » d''abord (rien n''est publié).' using errcode = 'P0061'; end if;
  update avis set suspendu_le = null, suspendu_motif = null where id = a.id;
  r := avis_message_commercant(a.id, ch, null, 'equipe');
  insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_REPONSE_APPLIQUEE',
    jsonb_build_object('avis', a.id, 'choix', ch, 'par', auth.jwt()->>'sub', 'le', now(), 'suspendu_le', a.suspendu_le, 'motif', a.suspendu_motif, 'resultat', r->>'choix'));
  return r || jsonb_build_object('applique_par', auth.jwt()->>'sub');
end $$;
revoke all on function admin_appliquer_reponse(uuid, text) from public, anon;
grant execute on function admin_appliquer_reponse(uuid, text) to authenticated;   -- vérifie est_admin()

-- Point 1 : « Débloquer » (console) : l'équipe a vérifié avec le commerçant. Contre-revue (E) : l'avis ne repart JAMAIS avec l'ancienne réponse :
-- la réponse reçue est oubliée (reponse_annoncee, heure, OK), l'attente reste, une tâche AVIS_A_RENVOYER demande à l'équipe d'écrire et
-- d'envoyer un texte (« Envoyer au commerçant » : nouvelle alerte, nouvelle heure) ; nouvelle référence. « Message lu » est refusé avant ce renvoi.
create or replace function admin_debloquer_faux(p_avis uuid) returns jsonb language plpgsql security definer set search_path = public as $$
declare a avis; r text;
begin
  if not est_admin() then raise exception 'réservé à l''administrateur' using errcode = 'P0005'; end if;
  select * into a from avis where id = p_avis for update;
  if a.id is null or a.faux_signale_le is null then raise exception 'Cet avis n''est pas signalé FAUX.' using errcode = 'P0061'; end if;
  update avis set faux_signale_le = null, reponse_annoncee = null, reponse_annoncee_le = null, publication_pas_avant = null, api_demande_le = null, api_motif = null,
                  decision_client = null, decision_client_le = null, retenue_le = coalesce(retenue_le, now()), attends_le = null, renvoi_attendu_le = null where id = a.id;
  delete from alertes_whatsapp where avis_id = a.id and type in ('avis_reponse','avis_rappel_message') and statut = 'a_envoyer';
  r := attribuer_ref_avis(a.id, a.commerce_id);
  update alertes_exploitant set vue_le = now() where type = 'AVIS_FAUX_SIGNALE' and detail->>'avis' = a.id::text and vue_le is null;
  insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_A_RENVOYER', a.commerce_id, jsonb_build_object('avis', a.id, 'note', a.note, 'ref', r, 'urgent', true));
  insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_FAUX_DEBLOQUE', jsonb_build_object('avis', a.id, 'par', 'equipe', 'signale_le', a.faux_signale_le, 'ref', r));
  return jsonb_build_object('id', a.id, 'debloque', true, 'ref', r, 'a_renvoyer', true);
end $$;

-- « Message lu » (corps du bloc « Attendez mon appel », recopié) : refusé tant qu'un avis débloqué après FAUX attend une nouvelle réponse.
create or replace function admin_liberer_avis(p_avis uuid) returns jsonb language plpgsql security definer set search_path = public as $$
declare a avis;
begin
  if not est_admin() then raise exception 'réservé à l''administrateur' using errcode = 'P0005'; end if;
  select * into a from avis where id = p_avis for update;
  if a.id is null then raise exception 'Avis introuvable.' using errcode = 'P0061'; end if;
  if a.retenue_le is null then return jsonb_build_object('id', a.id, 'retenue_le', null); end if;
  if exists (select 1 from alertes_exploitant e where e.vue_le is null and e.type = 'AVIS_A_RENVOYER' and e.detail->>'avis' = a.id::text) then
    raise exception 'Avis débloqué après FAUX : écrivez une nouvelle réponse et « Envoyer au commerçant » (l''ancienne ne repart jamais).' using errcode = 'P0061'; end if;
  update avis set retenue_le = null where id = a.id;
  update alertes_exploitant set vue_le = now() where type = 'AVIS_RETENU' and commerce_id = a.commerce_id and detail->>'avis' = a.id::text and vue_le is null;
  insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_LIBERE', jsonb_build_object('avis', a.id, 'retenue_le', a.retenue_le));
  return jsonb_build_object('id', a.id, 'retenue_le', null);
end $$;
revoke all on function admin_liberer_avis(uuid) from public, anon;
grant execute on function admin_liberer_avis(uuid) to authenticated;   -- vérifie est_admin()
-- admin_avis_publie (lot 10) : un avis signalé FAUX ne se publie pas, même à la main, avant « Débloquer » ; un avis suspendu non plus avant
-- « Appliquer OK / ATTENDS / FAUX » (contre-revue n°2) (remplacement d'un seul motif, rejouable).
do $l11p$
declare d text := pg_get_functiondef('admin_avis_publie(uuid, text, boolean)'::regprocedure);

begin

' || m);
  end if;

    execute replace(d, m, '  if a.suspendu_le is not null then raise exception ''Avis suspendu (message du commerçant à lire) : « Appliquer OK », « Appliquer ATTENDS » ou « Appliquer FAUX » d''''abord.'' using errcode = ''P0061''; end if;

-- ---------------------------------------------------------------- 8. Correction du commerçant réécrite par Claude (n8n « RENOM, avis : correction du commerçant »)
-- p_texte : la réponse réécrite et contrôlée ; p_refus : les règles non respectées (tableau), ou ['HUMAIN'] quand le message ne donne rien
-- d'utilisable ; p_version_le : version_client_le de la ligne sur laquelle n8n a travaillé (revue, point 4). Un message arrivé depuis rend cette
-- réécriture PÉRIMÉE : rien n'est gardé ni envoyé (le webhook du nouveau message relance avec toutes les corrections). 1 à 3 étoiles conforme : la
-- nouvelle réponse part au commerçant (alerte « avis_reponse », nouvelle heure), l'attente est levée ; si un message part déjà sur cet avis
-- (point 7), elle est mise en file juste après (renvoi_attendu_le, avis_publications_echues), jamais perdue ; avis sensible : brouillon pour
-- l'équipe (AVIS_CORRECTION_A_RELIRE) ; refus du contrôle ou sans WhatsApp : tâche AVIS_VERSION_CLIENT. Avis signalé FAUX : jamais réécrit.
create or replace function envoyer_reponse_corrigee(p_avis uuid) returns jsonb language plpgsql security definer set search_path = public as $$
declare a avis; ap timestamptz := apres_heures_calmes(now()); pa timestamptz;
begin
  select * into a from avis where id = p_avis for update;
  -- Contre-revue n°3 : avis suspendu ou signalé FAUX : RIEN ne part au commerçant ; la réécriture est un brouillon pour l'équipe (dans la tâche),
  -- jamais une « réponse reçue » (reponse_annoncee inchangée : « Appliquer OK » refuse un texte que le commerçant n'a pas reçu).
  if a.suspendu_le is not null or a.faux_signale_le is not null then
    update avis set renvoi_attendu_le = null where id = a.id;
    insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_VERSION_CLIENT', a.commerce_id,
      jsonb_build_object('avis', a.id, 'note', a.note, 'reecriture', case when a.faux_signale_le is not null then 'faux' else 'suspendu' end, 'brouillon', left(a.reponse_proposee, 2000)));
    return jsonb_build_object('etat', 'brouillon_equipe');
  end if;
  if exists (select 1 from alertes_whatsapp where avis_id = a.id and type = 'avis_reponse' and statut = 'en_cours') then
    update avis set renvoi_attendu_le = coalesce(renvoi_attendu_le, now()) where id = a.id;
    return jsonb_build_object('etat', 'renvoi_en_attente');
  end if;
  -- Contre-revue (A) : un ATTENDS reçu APRÈS le message que Claude vient d'intégrer : la réponse part sans heure de publication, l'attente reste
  -- (publiée 48 h après l'ATTENDS, sauf nouveau message : retenues_sans_message).
  if a.attends_le is not null and a.attends_le > coalesce(a.version_client_le, '-infinity'::timestamptz) then
    update avis set propose_le = now(), api_demande_le = null, publication_pas_avant = null, renvoi_attendu_le = null where id = a.id;
    delete from alertes_whatsapp where avis_id = a.id and type = 'avis_reponse';
    insert into alertes_whatsapp (commerce_id, avis_id, type, canal, prioritaire, envoyer_apres, donnees)
      values (a.commerce_id, a.id, 'avis_reponse', 'whatsapp', true, ap,
              jsonb_build_object('note', a.note, 'auteur', nullif(auteur_reduit(a.auteur), ''), 'reponse', left(a.reponse_proposee, 4096), 'texte', left(a.texte, 4096),
                                 'pas_avant', null, 'en_attente', true, 'ecrit_le', a.ecrit_le, 'sensible', false, 'echeance', a.cree_le + interval '24 hours', 'corrigee', true));
    return jsonb_build_object('etat', 'renvoyee_en_attente', 'attends_le', a.attends_le);
  end if;
  pa := avis_heure_annoncee(ap, a.cree_le);
  update avis set propose_le = now(), retenue_le = null, api_demande_le = null, publication_pas_avant = pa, renvoi_attendu_le = null where id = a.id;
  delete from alertes_whatsapp where avis_id = a.id and type = 'avis_reponse';
  insert into alertes_whatsapp (commerce_id, avis_id, type, canal, prioritaire, envoyer_apres, donnees)
    values (a.commerce_id, a.id, 'avis_reponse', 'whatsapp', true, ap,
            jsonb_build_object('note', a.note, 'auteur', nullif(auteur_reduit(a.auteur), ''), 'reponse', left(a.reponse_proposee, 4096), 'texte', left(a.texte, 4096),
                               'pas_avant', pa, 'ecrit_le', a.ecrit_le, 'sensible', false, 'echeance', a.cree_le + interval '24 hours', 'corrigee', true));
  return jsonb_build_object('etat', 'renvoyee', 'pas_avant', pa);
end $$;
revoke all on function envoyer_reponse_corrigee(uuid) from public, anon, authenticated;

drop function if exists avis_correction_redigee(uuid, text, jsonb);
create or replace function avis_correction_redigee(p_avis uuid, p_texte text, p_refus jsonb default '[]'::jsonb, p_version_le timestamptz default null) returns jsonb language plpgsql security definer set search_path = public as $$
declare a avis; c commerces; t text := nullif(btrim(coalesce(p_texte, '')), ''); r jsonb;
  refus jsonb := case when jsonb_typeof(p_refus) = 'array' then p_refus else '[]'::jsonb end;
begin
  select * into a from avis where id = p_avis for update;
  if a.id is null or a.statut not in ('escalade','humain') or a.version_client_le is null then return jsonb_build_object('etat', 'plus_en_attente'); end if;
  if a.faux_signale_le is not null then return jsonb_build_object('etat', 'faux'); end if;
  -- Contre-revue : sans p_version_le (ancien workflow pas réimporté), impossible de savoir si la réécriture est à jour : rien n'est envoyé, l'équipe reprend.
  if p_version_le is null then
    update avis set correction_redigee_le = now() where id = a.id;
    insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_VERSION_CLIENT', a.commerce_id,
      jsonb_build_object('avis', a.id, 'note', a.note, 'reecriture', 'refusee', 'refus', '["version du message absente : workflow de correction à réimporter"]'::jsonb));
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_CORRECTION_REFUSEE', jsonb_build_object('avis', a.id, 'refus', '["version absente"]'::jsonb));
    return jsonb_build_object('etat', 'equipe', 'refus', '["version du message absente"]'::jsonb);
  end if;
  if a.version_client_le > p_version_le then
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_CORRECTION_PERIMEE', jsonb_build_object('avis', a.id, 'version_le', p_version_le));
    return jsonb_build_object('etat', 'perimee');
  end if;
  if a.correction_redigee_le is not null and a.correction_redigee_le >= a.version_client_le then return jsonb_build_object('etat', 'deja'); end if;
  update avis set correction_redigee_le = now() where id = a.id;
  if t is not null and char_length(t) > 2000 then refus := refus || '["réponse trop longue"]'::jsonb; end if;
  select * into c from commerces where id = a.commerce_id;
  if t is null or jsonb_array_length(refus) > 0 then
    insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_VERSION_CLIENT', a.commerce_id, jsonb_build_object('avis', a.id, 'note', a.note, 'reecriture', 'refusee', 'refus', refus));
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_CORRECTION_REFUSEE', jsonb_build_object('avis', a.id, 'refus', refus));
    return jsonb_build_object('etat', 'equipe', 'refus', refus);
  end if;
  if coalesce(a.sensible, false) or a.statut = 'humain' then
    update avis set reponse_proposee = t where id = a.id;
    insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_CORRECTION_A_RELIRE', a.commerce_id, jsonb_build_object('avis', a.id, 'note', a.note, 'urgent', true));
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_CORRECTION_REDIGEE', jsonb_build_object('avis', a.id, 'sensible', true));
    return jsonb_build_object('etat', 'a_relire');
  end if;
  if not whatsapp_actif(c) then
    insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_VERSION_CLIENT', a.commerce_id, jsonb_build_object('avis', a.id, 'note', a.note, 'reecriture', 'sans_whatsapp'));
    return jsonb_build_object('etat', 'equipe', 'refus', '["numéro WhatsApp absent"]'::jsonb);
  end if;
  -- Contre-revue n°3 : avis suspendu : la réécriture reste un brouillon dans la tâche (réponse proposée inchangée), rien ne part.
  if a.suspendu_le is not null then
    insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_VERSION_CLIENT', a.commerce_id,
      jsonb_build_object('avis', a.id, 'note', a.note, 'reecriture', 'suspendu', 'brouillon', t));
    insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_CORRECTION_REDIGEE', jsonb_build_object('avis', a.id, 'sensible', false, 'etat', 'brouillon_equipe'));
    return jsonb_build_object('etat', 'brouillon_equipe');
  end if;
  update avis set reponse_proposee = t where id = a.id;
  update alertes_exploitant set vue_le = now() where vue_le is null and detail->>'avis' = a.id::text and type in ('AVIS_RETENU','AVIS_VERSION_CLIENT');
  r := envoyer_reponse_corrigee(a.id);
  insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_CORRECTION_REDIGEE', jsonb_build_object('avis', a.id, 'sensible', false, 'etat', r->>'etat', 'pas_avant', r->'pas_avant'));
  return r;
end $$;
revoke all on function avis_correction_redigee(uuid, text, jsonb, timestamptz) from public, anon, authenticated;
grant execute on function avis_correction_redigee(uuid, text, jsonb, timestamptz) to service_role;   -- n8n seulement

-- ---------------------------------------------------------------- 8 bis. ATTENDS sans message 48 h : la réponse reçue part par l'API (décision de PJ du 17/11/2026)
-- Corps du lot 10 recopié. Avis de 1 à 3 étoiles d'un commerce relié à l'API (avis reçu par l'API, fiche reliée) : 48 h après la demande d'attendre,
-- sans autre message, la réponse que le commerçant a REÇUE (même texte) part par l'API Google comme une publication à l'heure annoncée (même
-- règle avis_publiable_api, même workflow n8n), sans passer par « à publier à la main ». Commerce relevé par Places, ou réponse qui n'est plus
-- publiable par l'API (texte changé, nouvelle réponse en cours) : « à publier à la main », comme avant. Avis sensible : jamais publié d'office
-- (tâche urgente, inchangé) ; réponse jamais reçue : tâche urgente (inchangé).
create or replace function retenues_sans_message() returns jsonb language plpgsql security definer set search_path = public as $$
declare a avis; pub int := 0; urg int := 0; api int := 0;
begin
  for a in select * from avis where retenue_le is not null and traite_le is null and statut in ('escalade','humain')
             and faux_signale_le is null   -- revue, point 1 : un avis signalé FAUX n'est JAMAIS publié d'office (seule l'équipe le débloque)
             and suspendu_le is null   -- contre-revue n°2 : un avis suspendu n'est JAMAIS publié d'office (la suspension n'est pas un ATTENDS)
             and retenue_le <= now() - make_interval(hours => retenue_sans_message_heures())
             and (version_client_le is null or version_client_le < retenue_le) for update skip locked loop
    if coalesce(a.sensible, false) or a.statut = 'humain' or nullif(btrim(coalesce(a.reponse_proposee, '')), '') is null
       or a.reponse_annoncee_le is null or btrim(a.reponse_proposee) is distinct from btrim(coalesce(a.reponse_annoncee, '')) then
      if not exists (select 1 from alertes_exploitant where type = 'AVIS_SANS_MESSAGE_48H' and detail->>'avis' = a.id::text and detail->>'retenue_le' = to_jsonb(a.retenue_le)#>>'{}') then
        insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_SANS_MESSAGE_48H', a.commerce_id,
          jsonb_build_object('avis', a.id, 'note', a.note, 'sensible', coalesce(a.sensible, false), 'retenue_le', a.retenue_le, 'urgent', true));
        insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_SANS_MESSAGE_48H', jsonb_build_object('avis', a.id, 'sensible', coalesce(a.sensible, false), 'retenue_le', a.retenue_le));
        urg := urg + 1;
      end if;
    else
      -- levée : liberee_le = maintenant (déclencheur) ; l'heure annoncée est déjà passée (48 h après la demande), jamais avancée au-delà de maintenant.
      update avis set retenue_le = null, publication_pas_avant = least(coalesce(publication_pas_avant, now()), now()) where id = a.id;
      update alertes_exploitant set vue_le = now() where type = 'AVIS_RETENU' and commerce_id = a.commerce_id and detail->>'avis' = a.id::text and vue_le is null;
      if avis_publiable_api(a.id) = 'heure_annoncee' then
        perform demander_publication_api(a.id, 'attente_48h');
        insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_PUBLIE_48H_SANS_MESSAGE', jsonb_build_object('avis', a.id, 'retenue_le', a.retenue_le, 'par', 'api'));
        api := api + 1;
      else
        update avis set statut = 'a_publier_main' where id = a.id;
        insert into journal (commerce_id, type, detail) values (a.commerce_id, 'AVIS_PUBLIE_48H_SANS_MESSAGE', jsonb_build_object('avis', a.id, 'retenue_le', a.retenue_le, 'par', 'main'));
        pub := pub + 1;
      end if;
    end if;
  end loop;
  return jsonb_build_object('publies', pub + api, 'par_api', api, 'a_la_main', pub, 'urgents', urg);
end $$;
revoke all on function retenues_sans_message() from public, anon, authenticated;
grant execute on function retenues_sans_message() to service_role;

-- ---------------------------------------------------------------- 9. Passage régulier : heures annoncées échues, demandes perdues, corrections sans retour
create or replace function avis_publications_echues() returns jsonb language plpgsql security definer set search_path = public as $$
declare a record; n int := 0; r int := 0; k int := 0;
begin
  if not pg_try_advisory_xact_lock(hashtext('renom_avis_publications')) then return jsonb_build_object('deja', true); end if;
  for a in select id, api_demande_le from avis where statut = 'escalade' and publication_pas_avant <= now() and retenue_le is null
             and reponse_annoncee_le is not null and (api_demande_le is null or api_demande_le < now() - interval '30 minutes') loop
    if avis_publiable_api(a.id) = 'heure_annoncee' then perform demander_publication_api(a.id, 'heure_annoncee'); n := n + 1; end if;
  end loop;
  -- OK reçu dont la demande n'a pas abouti (n8n injoignable, résultat jamais rendu) : nouvelle demande après 30 minutes (PUT de Google : une
  -- seconde publication remplace la même réponse, sans doublon).
  for a in select id from avis where statut in ('escalade','humain') and api_motif = 'ok' and api_demande_le < now() - interval '30 minutes' loop
    if avis_publiable_api(a.id) = 'ok' then perform demander_publication_api(a.id, 'ok'); r := r + 1; end if;
  end loop;
  -- Contre-revue : OK reçu pendant une suspension (message non rattaché, non lu) : demandé dès que la suspension est levée.
  for a in select id from avis where statut in ('escalade','humain') and decision_client = 'approuve' and api_demande_le is null
             and decision_client_le >= coalesce(reponse_annoncee_le, 'infinity'::timestamptz) loop
    if avis_publiable_api(a.id) = 'ok' then perform demander_publication_api(a.id, 'ok'); r := r + 1; end if;
  end loop;
  -- Contre-revue n°2 : suspension ouverte depuis plus de 12 h : alerte urgente à l'équipe (une par suspension).
  for a in select x.id, x.commerce_id, x.note, x.suspendu_le, x.suspendu_motif from avis x where x.suspendu_le < now() - interval '12 hours' and x.statut in ('escalade','humain') loop
    if not exists (select 1 from alertes_exploitant e where e.type = 'AVIS_SUSPENDU_12H' and e.detail->>'avis' = a.id::text and e.detail->>'suspendu_le' = to_jsonb(a.suspendu_le)#>>'{}') then
      insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_SUSPENDU_12H', a.commerce_id,
        jsonb_build_object('avis', a.id, 'note', a.note, 'suspendu_le', a.suspendu_le, 'motif', a.suspendu_motif, 'urgent', true));
      k := k + 1;
    end if;
  end loop;
  -- Correction sans réponse de Claude après 30 minutes (n8n injoignable, Claude en panne) : tâche à l'équipe, une fois par message.
  for a in select x.id, x.commerce_id, x.note, x.version_client_le from avis x where x.statut in ('escalade','humain') and x.version_client_le < now() - interval '30 minutes'
             and x.version_client_le > now() - interval '7 days' and (x.correction_redigee_le is null or x.correction_redigee_le < x.version_client_le) loop
    if not exists (select 1 from alertes_exploitant e where e.type = 'AVIS_VERSION_CLIENT' and e.detail->>'avis' = a.id::text and e.cree_le >= a.version_client_le) then
      insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_VERSION_CLIENT', a.commerce_id, jsonb_build_object('avis', a.id, 'note', a.note, 'reecriture', 'sans_retour'));
      k := k + 1;
    end if;
  end loop;
  -- Point 7 : réponse corrigée qui attendait qu'un message en cours d'envoi soit parti : envoyée maintenant ; encore bloquée après 30 minutes :
  -- tâche à l'équipe (jamais de silence).
  for a in select x.id, x.commerce_id, x.note, x.renvoi_attendu_le from avis x where x.renvoi_attendu_le is not null and x.statut in ('escalade','humain') loop
    if not exists (select 1 from alertes_whatsapp w where w.avis_id = a.id and w.type = 'avis_reponse' and w.statut = 'en_cours') then
      perform envoyer_reponse_corrigee(a.id); k := k + 1;
    elsif a.renvoi_attendu_le < now() - interval '30 minutes'
          and not exists (select 1 from alertes_exploitant e where e.type = 'AVIS_VERSION_CLIENT' and e.detail->>'avis' = a.id::text and e.detail->>'reecriture' = 'renvoi_bloque' and e.vue_le is null) then
      insert into alertes_exploitant (type, commerce_id, detail) values ('AVIS_VERSION_CLIENT', a.commerce_id, jsonb_build_object('avis', a.id, 'note', a.note, 'reecriture', 'renvoi_bloque'));
    end if;
  end loop;
  return jsonb_build_object('demandes', n, 'relances', r, 'corrections_a_l_equipe', k);
end $$;
revoke all on function avis_publications_echues() from public, anon, authenticated;
grant execute on function avis_publications_echues() to service_role;

-- Toutes les 5 minutes si pg_cron est activé (Supabase Cron ; sans pg_cron, le passage horaire suffit). Marche aussi juste après
-- « create extension pg_cron » (appels en SQL dynamique : rien n'est lu dans le schéma cron avant). Rejouable sans doublon, quelle que soit la
-- version de pg_cron : toute tâche du même nom est retirée (cron.unschedule par son numéro), puis une seule est créée.
do $l11c$
declare j bigint;
begin
  if exists (select 1 from pg_extension where extname = 'pg_cron') or to_regprocedure('cron.schedule(text,text,text)') is not null then
    for j in execute $q$select jobid from cron.job where jobname = 'renom-avis-publication'$q$ loop
      execute 'select cron.unschedule($1)' using j;
    end loop;
    execute $c$select cron.schedule('renom-avis-publication', '*/5 * * * *', 'select public.avis_publications_echues()')$c$;
  end if;
end $l11c$;

-- Console, file des avis : où en est la publication automatique (« Publiée par l'API Google à partir du … », « en cours », « à publier à la main »).
create or replace function avis_publication_auto(p_id uuid) returns jsonb language plpgsql stable security definer set search_path = public as $$
declare a avis; c commerces; api boolean;
begin
  select * into a from avis where id = p_id;
  if a.id is null then return null; end if;
  select * into c from commerces where id = a.commerce_id;
  api := coalesce(a.source, 'api') <> 'places' and nullif(btrim(coalesce(c.google_location_id, '')), '') is not null;
  return jsonb_build_object('api', api, 'demandee_le', a.api_demande_le, 'prise_le', a.api_prise_le, 'motif', a.api_motif,
    'prevue_le', case when api and a.statut = 'escalade' and not coalesce(a.sensible, false) and a.retenue_le is null and a.reponse_annoncee_le is not null
                       and btrim(coalesce(a.reponse_proposee, '')) = btrim(coalesce(a.reponse_annoncee, '')) then a.publication_pas_avant end,
    'correction_redigee_le', a.correction_redigee_le, 'faux_signale_le', a.faux_signale_le, 'ref', a.ref_courte, 'renvoi_attendu_le', a.renvoi_attendu_le,
    'suspendue', a.suspendu_le is not null, 'suspendu_le', a.suspendu_le, 'suspendu_motif', a.suspendu_motif, 'attends_le', a.attends_le,
    'reponse_recue', case when a.reponse_annoncee_le is not null then a.reponse_annoncee end,
    'ok_possible', a.reponse_annoncee_le is not null and nullif(btrim(coalesce(a.reponse_annoncee, '')), '') is not null
                   and btrim(coalesce(a.reponse_proposee, '')) = btrim(a.reponse_annoncee),
    'a_renvoyer', exists (select 1 from alertes_exploitant e where e.vue_le is null and e.type = 'AVIS_A_RENVOYER' and e.detail->>'avis' = a.id::text),
    'corrections', jsonb_array_length(coalesce(a.corrections, '[]'::jsonb)));
end $$;
revoke all on function avis_publication_auto(uuid) from public, anon, authenticated;
do $l11f$
declare d text := pg_get_functiondef('admin_file_avis()'::regprocedure); m text := '''lien'', lien_avis_fiche(a, c))';
begin
  if position('avis_publication_auto' in d) = 0 and position(m in d) > 0 then
    execute replace(d, m, '''lien'', lien_avis_fiche(a, c), ''publication_auto'', avis_publication_auto(a.id), ''reponse_annoncee_le'', a.reponse_annoncee_le)');
  end if;
end $l11f$;

-- Déclencheurs vers n8n (même fonction renom_hook que webhooks_vault.sql, si elle existe déjà : sinon, coller webhooks_vault.sql après ce bloc).
do $l11h$
begin
  if exists (select 1 from pg_proc where proname = 'renom_hook' and pronamespace = 'public'::regnamespace) then
    execute 'drop trigger if exists hook_avis_publier on avis';
    execute $t$create trigger hook_avis_publier after update of api_demande_le on avis for each row
      when (new.api_demande_le is not null and new.api_demande_le is distinct from old.api_demande_le) execute function renom_hook('renom-avis-publier')$t$;
    execute 'drop trigger if exists hook_avis_correction on avis';
    execute $t$create trigger hook_avis_correction after update of version_client_le on avis for each row
      when (new.version_client_le is not null and new.version_client_le is distinct from old.version_client_le and new.statut in ('escalade','humain'))
      execute function renom_hook('renom-avis-correction')$t$;
  end if;
end $l11h$;

-- Types de journal du lot 11 : internes (jamais montrés au commerçant ; même liste que RENOM_JOURNAL_INTERNE de espace/web/donnees.js).
-- Dernière version (bloc « Relecture par l'équipe », 14/11/2026), recopiée, quatre types ajoutés.
create or replace function journal_interne(p_type text) returns boolean language sql immutable set search_path = public as $$
  select coalesce(starts_with(p_type, 'OBJECTIF_PILOTE_')          -- objectif interne des pilotes (ex-garantie « 10 avis »)
    or starts_with(p_type, 'GARANTIE_AVIS_')                        -- ancienne garantie « 10 avis » (bloc Modèle Vincent)
    or starts_with(p_type, 'DELAI_')                                -- garantie des délais : alertes « bientôt dépassé » à l'équipe
    or starts_with(p_type, 'EMAIL_ALERTE_')                         -- emails de la famille « exploitant » (alerte_mise_en_route, alerte_effacement)
    or starts_with(p_type, 'SECOURS_')                              -- mode de secours des avis (relevé Places, file à publier à la main), 20/10/2026
    or starts_with(p_type, 'PUBLICATION_A_LA_MAIN_')                -- publication à la main, sans Metricool (22/10/2026)
    or starts_with(p_type, 'WHATSAPP_')                             -- alertes WhatsApp : accord, retrait, envoi, échec (22/10/2026)
    or p_type in ('GARANTIE_DELAIS_SAISIE', 'MANQUEMENT_ECARTE', 'MOIS_OFFERT_A_APPLIQUER',   -- saisies et décisions de l'administrateur
                  'VALIDATION_TACITE_A_PROGRAMMER', 'MISE_EN_ROUTE_BLOQUEE', 'ERREUR_WORKFLOW', 'ECHEC_RELANCE_PROSPECT', 'RELEVE_ECHEC',
                  'AVIS_SENSIBLE_A_EFFACER', 'COMMENTAIRE_SENSIBLE_A_EFFACER', 'PAIEMENT_REFUSE', 'ABONNEMENT_SANS_COMMERCE',
                  'REMISE_LANCEMENT_DEJA_UTILISEE', 'ALERTE_PAR_EMAIL', 'ALERTE_EMAIL_ECHEC', 'ALERTE_SANS_CANAL',
                  'EFFACEMENT_AUTOMATIQUE',                         -- bilan de purger_donnees (bloc « Effacement automatique », 18/10/2026)
                  'CONTACT_REGLE', 'LIEN_AVIS_REGLE', 'AVIS_VERSION_CLIENT',   -- réglages de l'équipe, alerte de version (31/10/2026)
                  'AVIS_RETENU', 'AVIS_LIBERE',                     -- « Attendez mon appel » (01/11/2026)
                  'METRICOOL_MARQUE_NOTEE',                         -- blogId de la marque Metricool noté par la console (11/11/2026)
                  'AVIS_PUBLICATION_API_DEMANDEE', 'AVIS_CORRECTION_REDIGEE', 'AVIS_CORRECTION_REFUSEE', 'AVIS_CORRECTION_A_RELIRE',   -- lot 11 (17/11/2026)

                  'RELU_PAR_EQUIPE', 'LEGENDE_CORRIGEE_PAR_EQUIPE', 'CONTENU_A_REFAIRE', 'RELECTURE_EQUIPE_REGLEE'), false) $$;
revoke all on function journal_interne(text) from public, anon;
grant execute on function journal_interne(text) to authenticated, service_role;
